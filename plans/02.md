## 🎯 Зачем это нужно

Без доступа к документации агент будет:
- ❌ Генерировать код, противоречащий вашим стандартам
- ❌ Не знать о существующих утилитах и переиспользовать их
- ❌ Предлагать решения, которые вы уже пробовали и они не работают
- ❌ Нарушать архитектурные паттерны вашей компании

С документацией агент будет:
- ✅ Следовать вашим coding guidelines
- ✅ Использовать внутренние библиотеки и утилиты
- ✅ Избегать известных проблем (из post-mortem документов)
- ✅ Предлагать решения, соответствующие вашей архитектуре

## 📦 Реализация: Confluence Plugin

### 1. **Confluence Plugin для Semantic Kernel**

```csharp
public class ConfluencePlugin
{
    private readonly HttpClient _httpClient;
    private readonly string _baseUrl;
    private readonly string _apiToken;

    public ConfluencePlugin(string baseUrl, string email, string apiToken)
    {
        _baseUrl = baseUrl;
        _apiToken = Convert.ToBase64String(
            Encoding.ASCII.GetBytes($"{email}:{apiToken}"));
        
        _httpClient = new HttpClient();
        _httpClient.DefaultRequestHeaders.Authorization = 
            new AuthenticationHeaderValue("Basic", _apiToken);
    }

    [KernelFunction, Description("Search Confluence for runbooks and documentation")]
    public async Task<string> SearchDocumentationAsync(
        string query, 
        string spaceKey = null,
        int limit = 5)
    {
        var cql = $"type=page AND text~\"{query}\"";
        if (!string.IsNullOrEmpty(spaceKey))
        {
            cql += $" AND space=\"{spaceKey}\"";
        }

        var response = await _httpClient.GetAsync(
            $"{_baseUrl}/rest/api/content/search?cql={Uri.EscapeDataString(cql)}&limit={limit}");
        
        var result = await response.Content.ReadFromJsonAsync<ConfluenceSearchResult>();
        
        // Извлекаем контент страниц
        var pages = new List<string>();
        foreach (var page in result.Results)
        {
            var content = await GetPageContentAsync(page.Id);
            pages.Add($"# {page.Title}\n\n{content}\n");
        }

        return string.Join("\n---\n", pages);
    }

    [KernelFunction, Description("Get specific Confluence page by title")]
    public async Task<string> GetPageByTitleAsync(string title)
    {
        var response = await _httpClient.GetAsync(
            $"{_baseUrl}/rest/api/content?title={Uri.EscapeDataString(title)}&expand=body.storage");
        
        var result = await response.Content.ReadFromJsonAsync<ConfluenceSearchResult>();
        
        if (result.Results.Any())
        {
            return ConvertHtmlToMarkdown(result.Results.First().Body.Storage.Value);
        }
        
        return "Page not found";
    }

    [KernelFunction, Description("Get incident post-mortem from Confluence")]
    public async Task<string> GetPostMortemAsync(string incidentId)
    {
        // Ищем страницы с постмортемами
        var cql = $"type=page AND text~\"{incidentId}\" AND title~\"post-mortem\"";
        var response = await _httpClient.GetAsync(
            $"{_baseUrl}/rest/api/content/search?cql={Uri.EscapeDataString(cql)}");
        
        var result = await response.Content.ReadFromJsonAsync<ConfluenceSearchResult>();
        
        if (result.Results.Any())
        {
            var content = await GetPageContentAsync(result.Results.First().Id);
            return $"# Post-Mortem for {incidentId}\n\n{content}";
        }
        
        return "Post-mortem not found";
    }

    private async Task<string> GetPageContentAsync(string pageId)
    {
        var response = await _httpClient.GetAsync(
            $"{_baseUrl}/rest/api/content/{pageId}?expand=body.storage");
        
        var result = await response.Content.ReadFromJsonAsync<ConfluencePage>();
        return ConvertHtmlToMarkdown(result.Body.Storage.Value);
    }

    private string ConvertHtmlToMarkdown(string html)
    {
        // Используйте библиотеку типа HtmlAgilityPack + MarkdownConverter
        // или готовую типа ReverseMarkdown
        var converter = new ReverseMarkdown.Converter();
        return converter.Convert(html);
    }
}
```

### 2. **Jira Plugin (для связи с тикетами)**

```csharp
public class JiraPlugin
{
    [KernelFunction, Description("Get related Jira tickets for an incident")]
    public async Task<string> GetRelatedTicketsAsync(string componentName, string errorType)
    {
        var jql = $"component = \"{componentName}\" AND text ~ \"{errorType}\" ORDER BY created DESC";
        
        var response = await _jiraClient.SearchAsync(jql, maxResults: 10);
        
        var tickets = response.Issues.Select(issue => new
        {
            Key = issue.Key,
            Summary = issue.Fields.Summary,
            Status = issue.Fields.Status.Name,
            Description = issue.Fields.Description,
            Comments = issue.Fields.Comment.Comments
                .Select(c => c.Body)
                .Take(3) // Последние 3 комментария
        });

        return JsonSerializer.Serialize(tickets, new JsonSerializerOptions 
        { 
            WriteIndented = true 
        });
    }

    [KernelFunction, Description("Get resolution from similar past incidents")]
    public async Task<string> GetPastResolutionsAsync(string errorMessage)
    {
        // Ищем закрытые тикеты с похожими ошибками
        var jql = $"resolution = Fixed AND text ~ \"{errorMessage}\" ORDER BY resolved DESC";
        
        var response = await _jiraClient.SearchAsync(jql, maxResults: 5);
        
        var resolutions = response.Issues.Select(issue => new
        {
            Ticket = issue.Key,
            Summary = issue.Fields.Summary,
            Resolution = issue.Fields.Resolution?.Name,
            FixVersion = issue.Fields.FixVersions?.FirstOrDefault()?.Name,
            // Извлекаем описание фикса из комментариев
            FixDescription = ExtractFixDescription(issue)
        });

        return JsonSerializer.Serialize(resolutions, new JsonSerializerOptions 
        { 
            WriteIndented = true 
        });
    }
}
```

### 3. **Git Wiki / README Plugin**

```csharp
public class GitDocsPlugin
{
    [KernelFunction, Description("Search project documentation in Git repository")]
    public async Task<string> SearchProjectDocsAsync(string query)
    {
        var docsFolder = Path.Combine(_repoPath, "docs");
        var readmePath = Path.Combine(_repoPath, "README.md");
        
        var results = new List<string>();
        
        // Ищем в docs/
        if (Directory.Exists(docsFolder))
        {
            var markdownFiles = Directory.GetFiles(docsFolder, "*.md", SearchOption.AllDirectories);
            foreach (var file in markdownFiles)
            {
                var content = await File.ReadAllTextAsync(file);
                if (content.Contains(query, StringComparison.OrdinalIgnoreCase))
                {
                    results.Add($"# {Path.GetFileName(file)}\n\n{content}");
                }
            }
        }
        
        // Добавляем README
        if (File.Exists(readmePath))
        {
            var readme = await File.ReadAllTextAsync(readmePath);
            results.Insert(0, $"# README.md\n\n{readme}");
        }
        
        return string.Join("\n---\n", results.Take(5));
    }
}
```

## 🧠 RAG (Retrieval-Augmented Generation) для больших объемов документации

Если у вас много документации (сотни страниц в Confluence), лучше использовать **векторный поиск**:

### 1. **Индексация документации**

```csharp
public class DocumentationIndexer
{
    private readonly IEmbeddingGeneration<string, float> _embeddingGenerator;
    private readonly QdrantVectorDbClient _vectorDb;

    public async Task IndexConfluenceSpaceAsync(string spaceKey)
    {
        var pages = await _confluencePlugin.GetAllPagesAsync(spaceKey);
        
        foreach (var page in pages)
        {
            // Разбиваем на чанки по 500 токенов
            var chunks = ChunkText(page.Content, maxTokens: 500);
            
            foreach (var chunk in chunks)
            {
                // Генерируем эмбеддинг
                var embedding = await _embeddingGenerator.GenerateEmbeddingAsync(chunk);
                
                // Сохраняем в векторную БД
                await _vectorDb.UpsertAsync("confluence_docs", new VectorRecord
                {
                    Id = Guid.NewGuid().ToString(),
                    Vector = embedding,
                    Metadata = new Dictionary<string, object>
                    {
                        ["title"] = page.Title,
                        ["url"] = page.Url,
                        ["chunk"] = chunk,
                        ["indexed_at"] = DateTime.UtcNow
                    }
                });
            }
        }
    }

    private List<string> ChunkText(string text, int maxTokens)
    {
        // Используем токенизатор для точного разбиения
        var tokenizer = new GPT2Tokenizer();
        var tokens = tokenizer.Encode(text);
        
        var chunks = new List<string>();
        for (int i = 0; i < tokens.Length; i += maxTokens)
        {
            var chunkTokens = tokens.Skip(i).Take(maxTokens).ToArray();
            chunks.Add(tokenizer.Decode(chunkTokens));
        }
        
        return chunks;
    }
}
```

```csharp
public class DocumentationRetriever
{
    [KernelFunction, Description("Find relevant documentation for the current incident")]
    public async Task<string> FindRelevantDocsAsync(
        string errorMessage, 
        string componentName,
        string metricName)
    {
        // Формируем поисковый запрос
        var query = $"{errorMessage} {componentName} {metricName}";
        
        // Генерируем эмбеддинг запроса
        var queryEmbedding = await _embeddingGenerator.GenerateEmbeddingAsync(query);
        
        // Ищем ближайшие векторы
        var results = await _vectorDb.SearchAsync(
            "confluence_docs", 
            queryEmbedding, 
            topK: 5);
        
        // Форматируем результаты
        var docs = results.Select(r => new
        {
            Title = r.Metadata["title"],
            Url = r.Metadata["url"],
            Content = r.Metadata["chunk"]
        });
        
        return JsonSerializer.Serialize(docs, new JsonSerializerOptions 
        { 
            WriteIndented = true 
        });
    }
}
```

```csharp
public class SelfHealingAgent
{
    public async Task<HealingResult> AnalyzeAndFixAsync(AlertContext alert)
    {
        // Шаг 1: Сбор технического контекста
        var metrics = await _prometheusPlugin.GetMetricsAsync(alert.MetricName);
        var logs = await _lokiPlugin.GetErrorLogsAsync(alert.ServiceName);
        
        // Шаг 2: Поиск документации (НОВОЕ!)
        var relevantDocs = await _docRetriever.FindRelevantDocsAsync(
            alert.ErrorMessage, 
            alert.ComponentName,
            alert.MetricName);
        
        var postMortems = await _confluencePlugin.GetPostMortemAsync(alert.IncidentId);
        var pastResolutions = await _jiraPlugin.GetPastResolutionsAsync(alert.ErrorMessage);
        
        // Шаг 3: Анализ с учетом документации
        var analysisResult = await _kernel.InvokeAsync<string>(
            "AnalysisAgent",
            "AnalyzeIncident",
            new()
            {
                ["metrics"] = metrics,
                ["logs"] = logs,
                ["documentation"] = relevantDocs,
                ["past_incidents"] = postMortems,
                ["past_resolutions"] = pastResolutions,
                ["code_structure"] = await _codeAnalyzer.AnalyzeProjectAsync(_projectPath)
            });

        // Шаг 4: Генерация плана с учетом стандартов компании
        var codingStandards = await _confluencePlugin.GetPageByTitleAsync("C# Coding Standards");
        
        var plan = await _kernel.InvokeAsync<FixPlan>(
            "PlanningAgent",
            "CreateFixPlan",
            new()
            {
                ["analysis"] = analysisResult,
                ["coding_standards"] = codingStandards,
                ["architecture_docs"] = await _gitDocsPlugin.SearchProjectDocsAsync("architecture")
            });

        // ... остальная логика (применение фикса, тесты, коммит)
    }
}
```

## 📋 Какие источники можно подключить

| Источник | Что дает | Как использовать |
|----------|----------|------------------|
| **Confluence** | Runbooks, архитектурные решения, post-mortem | RAG + прямой поиск |

| **SharePoint** | Корпоративные стандарты, политики | RAG |

| **Slack/Teams** | Обсуждения инцидентов, быстрые решения | Поиск по каналам |
| **Stack Overflow (внутренний)** | Внутренняя база знаний | RAG |
| **Swagger/OpenAPI** | API контракты, схемы данных | Прямое чтение |
| **Database Schema** | Структура БД, индексы | Прямой запрос к INFORMATION_SCHEMA |

### Background Job для индексации

```csharp
public class DocumentationSyncJob : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                // Синхронизируем Confluence каждые 6 часов
                await _indexer.IndexConfluenceSpaceAsync("RUNBOOKS");
                await _indexer.IndexConfluenceSpaceAsync("ARCHITECTURE");
                
                // Синхронизируем Jira каждые 2 часа
                await _indexer.IndexJiraTicketsAsync("component = Backend");
                
                _logger.LogInformation("Documentation sync completed");
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Documentation sync failed");
            }
            
            await Task.Delay(TimeSpan.FromHours(2), stoppingToken);
        }
    }
}
```

## 🔐 Безопасность и доступы

1. **Service Account:** Создайте отдельного пользователя для агента с read-only доступом к Confluence/Jira
2. **API Tokens:** Используйте API tokens вместо паролей
3. **Scope:** Ограничьте доступ только к нужным пространствам (spaces)
4. **Audit Log:** Логируйте все запросы к документации
5. **Secrets Management:** Храните токены в Azure Key Vault / AWS Secrets Manager

## 💡 Практические советы

1. **Начните с Runbooks:** Сначала подключите только runbooks (инструкции по решению типичных проблем). Это даст максимальный ROI.

2. **Помечайте документацию:** Добавьте в Confluence метки (labels) типа `ai-agent`, `runbook`, `architecture`, чтобы агент мог фильтровать релевантные документы.

3. **Структурируйте post-mortem:** Создайте шаблон для post-mortem в Confluence с обязательными полями: Root Cause, Resolution, Prevention. Агенту будет легче парсить.

4. **Кэшируйте:** Не запрашивайте одну и ту же страницу Confluence каждый раз. Кэшируйте на 1-2 часа.

5. **Fallback:** Если документация не найдена, агент должен работать как раньше (без контекста).

6. **Версионирование:** Если вы обновили документацию, убедитесь, что индекс обновился (через background job).
